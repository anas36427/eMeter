        \draw [line] (backend) -- (uc_cache);
        \draw [line] (backend) -- (uc_sync);

        % Includes and Extends Relationships
        
        % Capturing reading includes searching the consumer first
        \draw [include] (uc_capture) -- node[right, font=\scriptsize] {<<include>>} (uc_search);
        
        % Caching data and Searching data requires authentication
        \draw [include] (uc_cache) -- node[right, font=\scriptsize] {<<include>>} (uc_login);

        \draw [include] (uc_capture) -- node[above, font=\scriptsize] {<<include>>} (uc_validate);
        
        % Queueing locally extends the capture process (happens when offline)
        \draw [extend] (uc_capture) -- node[right, font=\scriptsize] {<<extend>>} (uc_queue);
        
        % Syncing to cloud notifies the user
        \draw [include] (uc_sync) -- node[above, font=\scriptsize] {<<include>>} (uc_notify);
